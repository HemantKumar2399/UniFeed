+import { useRef, useState } from "react";

+interface LongPressButtonProps
+  extends React.ButtonHTMLAttributes<HTMLButtonElement> {
+  onComplete: () => void;
+  duration?: number;
+  progressClassName?: string;
+}
+
+export function LongPressButton({
+  onComplete,
+  duration = 1500,
+  className,
+  progressClassName,
+  children,
+  ...props
+}: LongPressButtonProps) {
+  const [isPressing, setIsPressing] = useState(false);
+  const timerRef = useRef<NodeJS.Timeout | null>(null);
+
+  const startPress = (e: React.PointerEvent<HTMLButtonElement>) => {
+    // Prevent default touch behaviors like magnifying glass on iOS
+    e.preventDefault();
+
+    if (e.pointerType === "mouse" && e.button !== 0) return;
+
+    setIsPressing(true);
+    timerRef.current = setTimeout(() => {
+      setIsPressing(false);
+      onComplete();
+    }, duration);
+  };
+
+  const cancelPress = () => {
+    setIsPressing(false);
+    if (timerRef.current) {
+      clearTimeout(timerRef.current);
+      timerRef.current = null;
+    }
+  };
+
+  return (
+    <button
+      {...props}
+      onPointerDown={startPress}
+      onPointerUp={cancelPress}
+      onPointerLeave={cancelPress}
+      onPointerCancel={cancelPress}
+      onContextMenu={(e) => e.preventDefault()}
+      className={cn(
+        "relative overflow-hidden select-none touch-none group",
+        "transition-transform active:scale-95",
+        "-webkit-touch-callout-none", 
+        className
+      )}
+      style={{
+        WebkitTouchCallout: "none",
+        WebkitUserSelect: "none",
+        userSelect: "none",
+      }}
+      aria-pressed={isPressing}
+    >
+      <div
+        className={cn(
+          "absolute left-0 top-0 bottom-0 bg-rose-600/30 opacity-0 pointer-events-none will-change-[width]",
+          progressClassName
+        )}
+        style={{
+          width: isPressing ? "100%" : "0%",
+          transition: isPressing
+            ? `width ${duration}ms linear, opacity 100ms ease-in`
+            : "width 0ms, opacity 200ms ease-out",
+          opacity: isPressing ? 1 : 0,
+        }}

+import { useState } from "react";
+import { api } from "@/trpc/react";
+
+export function CapacityBar({ eventId, initialCount, totalCapacity }: { eventId: number, initialCount: number, totalCapacity: number }) {
+  const [currentCount, setCurrentCount] = useState(initialCount);
+
+  // Subscribe to capacity changes
+  api.rsvp.onCapacityChange.useSubscription(
+    { eventId },
+    {
+      onData: (data: { eventId: number; newCapacity: number }) => {
+        // data.newCapacity is the remaining capacity
+        setCurrentCount(totalCapacity - data.newCapacity);
+      },
+    }
+  );
+
+  const percentage = Math.min(100, Math.max(0, (currentCount / totalCapacity) * 100));
+
+  let barColorClass = "bg-zinc-300";
+  if (percentage >= 95) {
+    barColorClass = "bg-rose-600";
+  } else if (percentage >= 75) {
+    barColorClass = "bg-amber-500";
+  }
+
+  return (
+    <div className="w-full bg-zinc-800 rounded-full h-1.5 mt-2 overflow-hidden">
+      <div
+        className={`h-full ${barColorClass} transition-all duration-300 ease-out`}
+        style={{ width: `${percentage}%` }}
+      />
+    </div>
+  );
+}

+import { observable } from "@trpc/server/observable";
+
+export const rsvpRouter = createTRPCRouter({
+  onCapacityChange: publicProcedure
+    .input(z.object({ eventId: z.number().int().positive() }))
+    .subscription(({ input }) => {
+      return observable<{ eventId: number; newCapacity: number }>((emit) => {
+        const onCapacityChange = (data: { eventId: number; newCapacity: number }) => {
+          if (data.eventId === input.eventId) {
+            emit.next(data);
+          }
+        };
+
+        ee.on("capacityChange", onCapacityChange);
+
+        return () => {
+          ee.off("capacityChange", onCapacityChange);
+        };
+      });
+    }),
+
+  register: protectedProcedure
+    .input(z.object({ eventId: z.number().int().positive() }))
+    .mutation(async ({ ctx, input }) => {
+      const userId = ctx.session.user.id;
+
+      try {
+        const { rsvp, remainingCapacity } = await createRsvpPessimistic(userId, input.eventId);
+        ee.emit("capacityChange", { eventId: input.eventId, newCapacity: remainingCapacity });
+        return rsvp;
+      } catch (error: unknown) {
+        if (error instanceof CapacityError) {
+          throw new TRPCError({
+            code: "PRECONDITION_FAILED",
+            message: "EVENT_FULL",
+          });
+        }
+
+        // Prisma unique constraint violation code for `@@unique([userId, eventId])`
+        if (error && typeof error === "object" && "code" in error && error.code === "P2002") {
+          throw new TRPCError({
+            code: "CONFLICT",
+            message: "ALREADY_REGISTERED",
+          });
+        }
+
+        throw new TRPCError({
+          code: "INTERNAL_SERVER_ERROR",
+          message: "An unexpected error occurred during RSVP.",
+        });
+      }

+import { db } from "@/server/db";
+import { Prisma } from "../../../generated/prisma";
+
+export class CapacityError extends Error {
+  constructor(message: string) {
+    super(message);
+    this.name = "CapacityError";
+  }
+}
+
+export async function createRsvpPessimistic(userId: string, eventId: number) {
+  // Use $transaction to ensure atomicity
+  return await db.$transaction(async (tx) => {
+    // Pessimistic lock on the event row: SELECT ... FOR UPDATE
+    const eventParams = Prisma.sql`
+      SELECT id, capacity, "createdAt", "updatedAt"
+      FROM "Event"
+      WHERE id = ${eventId}
+      FOR UPDATE
+    `;
+
+    // Execute raw query for the lock
+    const events = await tx.$queryRaw<
+      { id: number; capacity: number }[]
+    >(eventParams);
+
+    if (!events || events.length === 0) {
+      throw new Error("Event not found");
+    }
+
+    const event = events[0];
+
+    // Check how many RSVPs currently exist
+    const rsvpCount = await tx.rSVP.count({
+      where: { eventId },
+    });
+
+    if (!event) {
+      throw new Error("Event not found");
+    }
+
+    const remainingCapacity = event.capacity - rsvpCount;
+    if (remainingCapacity <= 0) {
+      throw new CapacityError("Event is at full capacity");
+    }
+
+    // Insert the RSVP (will fail safely if the unique constraint is blown)
+    const newRsvp = await tx.rSVP.create({
+      data: {
+        userId,
+        eventId,
+      },
+    });
+
+    return { rsvp: newRsvp, remainingCapacity: remainingCapacity - 1 };
+  });
+}
